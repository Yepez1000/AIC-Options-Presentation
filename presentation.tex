\begin{frame}{What Is an Option?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \textbf{An option} gives its holder the \alert{right, not the obligation}, to transact in an underlying asset on specified terms.
      \vspace{0.35cm}
      \begin{itemize}
        \item \textbf{Call:} right to buy at the strike price $K$.
        \item \textbf{Put:} right to sell at the strike price $K$.
        \item \textbf{Expiration $T$:} last date the right can be exercised.
        \item \textbf{Premium:} price paid today for the contract.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{block}{Central question}
        How much should someone pay today for a claim that provides a potentially uncertain payoff in the future?
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{An Arbitrary Financial Claim}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{tikzpicture}[x=1cm,y=1cm,>=stealth]
        \node (zero) at (0,0) {$S_0,\ V_0$};
        \node (up) at (3,1.35) {$S_u,\ V_u$};
        \node (down) at (3,-1.35) {$S_d,\ V_d$};
        \draw[->,thick] (zero) -- (up);
        \draw[->,thick] (zero) -- (down);
        \node[aggieGray] at (1.45,0.95) {up state};
        \node[aggieGray] at (1.45,-0.95) {down state};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      Suppose a stock trades today at $S_0$. At time $T$, it can be $S_u$ or $S_d$.
      \vspace{0.25cm}
      A claim $V$ written on the stock then pays $V_u$ or $V_d$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Replicating the Financial Claim}
  Rather than price the claim directly, build a portfolio with exactly the same payoff in every state.
  \vspace{0.3cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      Hold $\Delta$ shares of stock and borrow $y$ dollars today at continuously compounded rate $r$.
      \vspace{0.2cm}
      \small
      $\Delta$ is the shares you have bought; $y$ is the amount of the loan.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{align*}
        V_u &= \Delta S_u - y e^{rT},\\
        V_d &= \Delta S_d - y e^{rT}.
      \end{align*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Solving for $\Delta$ and the Loan}
  \begin{align*}
    V_u - V_d &= \Delta(S_u-S_d) \\[0.15cm]
    \therefore\qquad \Delta &= \frac{V_u-V_d}{S_u-S_d}\\[0.15cm]
    y &= e^{-rT}(\Delta S_d-V_d)
  \end{align*}
  \begin{center}
    $\Delta$ is the shares you have bought. \hspace{1cm} $y$ is the amount of the loan.
  \end{center}
\end{frame}

\begin{frame}{No-Arbitrage: The Law of One Price}
  \begin{block}{Portfolio lemma}
    If two portfolios have identical future payoffs in every state, then an arbitrage-free market must assign them the same price today.
  \end{block}
  \vspace{0.25cm}
  \begin{center}
    \Large $\displaystyle V_0 = \Delta S_0-y$
  \end{center}
\end{frame}

\begin{frame}{A One-Step Call Option Example}
  \small
  $S_0=\$100$, $S_u=\$120$, $S_d=\$80$, $K=\$100$, and $r=0$. Therefore,
  \begin{align*}
    V_u&=\max(120-100,0)=\$20, & V_d&=\max(80-100,0)=\$0,\\
    \Delta&=\frac{20-0}{120-80}=0.5, & y&=0.5(80)-0=\$40,\\
    V_0&=0.5(100)-40=\boxed{\$10}.
  \end{align*}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Up state}
        Half a share is worth $\$60$. Repay $\$40$. Remaining payoff: $\$20$.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Down state}
        Half a share is worth $\$40$. Repay $\$40$. Remaining payoff: $\$0$.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \centering\textbf{The $\$10$ premium is paid today; $\$20$ or $\$0$ is paid at expiration.}
\end{frame}

\begin{frame}{From Replication to Risk-Neutral Pricing}
  Insert the replicating portfolio into $V_0=\Delta S_0-y$:
  \begin{align*}
    V_0 &= \Delta S_0-e^{-rT}(\Delta S_d-V_d)\\
        &= e^{-rT}\left[qV_u+(1-q)V_d\right],
  \end{align*}
  where
  \begin{center}
    \Large $\displaystyle q=\frac{S_0e^{rT}-S_d}{S_u-S_d}$.
  \end{center}
\end{frame}

\begin{frame}{What Are Risk-Neutral Probabilities?}
  \begin{center}
    \Large $\displaystyle S_0=e^{-rT}\left[qS_u+(1-q)S_d\right]$
  \end{center}
  The weights $q$ and $1-q$ make the stock's expected return equal to the risk-free rate.
  \vspace{0.25cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Actual probabilities}
        Describe beliefs about what will happen in the world.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Risk-neutral probabilities}
        Are pricing weights consistent with traded-asset prices and no-arbitrage.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{block}{No-arbitrage condition}
    $S_d<S_0e^{rT}<S_u$ ensures $0<q<1$. These weights arise from replication, not from forecasting the next stock move.
  \end{block}
\end{frame}

\begin{frame}{Extending to Multiple Time Steps}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{tikzpicture}[x=1.05cm,y=0.63cm,>=stealth,scale=0.8,every node/.style={font=\scriptsize}]
        \foreach \x/\y in {0/0,1/1,1/-1,2/2,2/0,2/-2,3/3,3/1,3/-1,3/-3}
          \fill[aggieMaroon] (\x,\y) circle (1.5pt);
        \foreach \a/\b/\c/\d in {0/0/1/1,0/0/1/-1,1/1/2/2,1/1/2/0,1/-1/2/0,1/-1/2/-2,2/2/3/3,2/2/3/1,2/0/3/1,2/0/3/-1,2/-2/3/-1,2/-2/3/-3}
          \draw[gray] (\a,\b)--(\c,\d);
        \node[below] at (0,-0.25) {$0$};
        \node[below] at (1,-1.25) {$\Delta t$};
        \node[below] at (2,-2.25) {$2\Delta t$};
        \node[below] at (3,-3.25) {$T$};
        \node[right] at (3,3) {terminal payoff};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.46\textwidth}
      At every terminal node, the contract tells us the payoff. Then work backward:
      \begin{align*}
        V_t=e^{-r\Delta t}\big[qV_{t+\Delta t}^{u}+(1-q)V_{t+\Delta t}^{d}\big].
      \end{align*}
      \small
      \begin{enumerate}
        \item Calculate terminal payoffs.
        \item Weight and discount adjacent values.
        \item Repeat until time zero.
      \end{enumerate}
    \end{column}
  \end{columns}
  \centering\small Terminal nodes are model-generated scenarios, not forecasts of realized prices.
\end{frame}

\begin{frame}{From Binomial Trees to Black--Scholes}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \begin{block}{Discrete binomial model}
        Stock prices move up or down at individual time steps.
      \end{block}
      \centering\Large More steps, smaller $\Delta t$
    \end{column}
    \begin{column}{0.08\textwidth}
      \vspace{0.7cm}\centering\Large $\rightarrow$
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Continuous-time model}
        Stock prices evolve continuously under a stochastic process.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.35cm}
  Under an appropriate parameterization, the binomial price converges to the Black--Scholes--Merton price as the number of steps grows.
\end{frame}

\begin{frame}{Black--Scholes--Merton: Equation and Price}
  \small
  Continuous-time delta hedging and no-arbitrage lead to
  \begin{align*}
    \frac{\partial V}{\partial t}+\frac{1}{2}\sigma^2S^2\frac{\partial^2V}{\partial S^2}+rS\frac{\partial V}{\partial S}-rV=0.
  \end{align*}
  Solving this equation gives the price of the call option:
  \begin{center}
    \Large $\displaystyle V_0=e^{-rT}\mathbb{E}^{\mathbb{Q}}[\operatorname{Payoff}(S_T)].$
  \end{center}
  \small For a call, $\operatorname{Payoff}(S_T)=\max(S_T-K,0)$, where $S_T$ comes from the assumed asset-price model.
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      For a European call with no dividends,
      \begin{align*}
        C&=S_0N(d_1)-Ke^{-rT}N(d_2),\\
        d_1&=\frac{\ln(S_0/K)+(r+\sigma^2/2)T}{\sigma\sqrt{T}},\quad d_2=d_1-\sigma\sqrt{T}.
      \end{align*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What Does Black--Scholes Actually Tell Us?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{It calculates}
        A theoretical arbitrage-free option premium under its assumptions.
      \end{block}
      \vspace{0.25cm}
      \begin{align*}
        C_T&=\max(S_T-K,0),\\
        P_T&=\max(K-S_T,0).
      \end{align*}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{It does not calculate}
        A prediction of the realized stock price at expiration.
      \end{block}
      \vspace{0.25cm}
      Terminal payoff depends on the unknown $S_T$. A traded market premium can differ from the model price.
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{block}{Keep the terms separate}
    The premium is the upfront price of the contract, not a fee added to its theoretical value.
  \end{block}
\end{frame}

\begin{frame}{The Greeks: Sensitivities of Option Value}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Delta: $\frac{\partial V}{\partial S}$}
        Sensitivity to stock price; locally, the shares needed to hedge the option.
      \end{block}
      \begin{block}{Gamma: $\frac{\partial^2 V}{\partial S^2}$}
        Sensitivity of Delta to the stock price.
      \end{block}
      \begin{block}{Theta}
        Sensitivity to time; its sign depends on whether time is calendar time or time to expiration.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Vega: $\frac{\partial V}{\partial \sigma}$}
        Sensitivity to volatility.
      \end{block}
      \begin{block}{Rho: $\frac{\partial V}{\partial r}$}
        Sensitivity to the interest rate.
      \end{block}
      \vspace{0.25cm}
      \textbf{Connection:} the $\Delta$ from the one-step replicating portfolio becomes a local, continuously adjusted hedge ratio.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Volatility Smirk: Why Downside Skew Exists}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[x=0.78cm,y=0.72cm]
        \draw[->,thick] (0,0) -- (6.2,0) node[right,font=\scriptsize] {Strike};
        \draw[->,thick] (0,0) -- (0,4.6) node[above,font=\scriptsize] {IV};
        \draw[densely dotted,aggieGray,thick] (0.35,2.15) -- (5.85,2.15);
        \draw[dashed,blue!65!black,thick,smooth] plot coordinates
          {(0.35,3.15) (1.2,2.65) (2.2,2.25) (3.1,2.05) (4.1,2.25) (5.0,2.65) (5.85,3.15)};
        \draw[aggieMaroon,very thick,smooth] plot coordinates
          {(0.35,4.05) (1.0,3.45) (1.8,2.85) (2.7,2.35) (3.6,2.05) (4.7,1.85) (5.85,1.75)};
        \draw[densely dashed,black!45] (3.1,0) -- (3.1,4.25);
        \node[below,font=\scriptsize] at (1.0,0) {Downside};
        \node[below,font=\scriptsize] at (3.1,0) {ATM};
        \node[below,font=\scriptsize] at (5.2,0) {Upside};
        \draw[densely dotted,aggieGray,thick] (0.55,-0.85) -- (1.15,-0.85)
          node[right,black,font=\scriptsize] {Flat};
        \draw[dashed,blue!65!black,thick] (2.05,-0.85) -- (2.65,-0.85)
          node[right,black,font=\scriptsize] {Smile};
        \draw[aggieMaroon,very thick] (3.75,-0.85) -- (4.35,-0.85)
          node[right,black,font=\scriptsize] {Smirk};
      \end{tikzpicture}
      \vspace{0.12cm}

      {\scriptsize Same expiration: implied volatility across strikes.}
    \end{column}
    \begin{column}{0.51\textwidth}
      \small
      \begin{block}{An asymmetric curve}
        Flat skew assigns similar IV across strikes. A smile raises IV in both tails. A \textbf{smirk} is steeper on one side of the distribution than the other.
      \end{block}
      \begin{block}{What downside smirk conveys}
        Market prices assign a larger effect on implied volatility to a downside move than to a comparable upside move. Lower-strike options therefore trade at higher IV.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.08cm}
  \scriptsize
  \begin{block}{How does skew exist? Supply meets demand}
    Investors buy puts to protect portfolios against falling markets, which have historically produced larger and more severe moves. That demand raises downside-option prices and their implied volatilities. At a fundamental level, skew emerges as buyers and sellers meet at different prices across strikes; implied volatility is simply those option prices expressed in volatility terms.
  \end{block}
\end{frame}

\begin{frame}{Why Implied Volatility Often Exceeds Realized Volatility}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.59\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=5.85cm,keepaspectratio]{Impliedvsrealizedvol.png}
      {\scriptsize S\&P 500, Jan. 2025--Apr. 2026. A negative IV spread means implied volatility exceeded trailing realized volatility.}
    \end{column}
    \begin{column}{0.38\textwidth}
      \scriptsize
      \begin{block}{Demand for options}
        Traders pay a premium for efficient, leveraged exposure to the underlying asset and, more importantly, protection against adverse price movements.
      \end{block}
      \begin{block}{Compensation for sellers}
        Option sellers are the counterparties to that demand. They require compensation for bearing volatility risk and the possibility of extreme tail losses.
      \end{block}
      \begin{block}{The volatility risk premium}
        Together, option demand and seller compensation contribute to implied volatility often exceeding subsequent realized volatility. Academics call this difference the \textbf{volatility risk premium}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Iron Condor: Strike Math and Payoff}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.51\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=4.0cm,keepaspectratio]{Iron Condor.png}
      \vspace{0.03cm}
      {\scriptsize SPY example shown: \$771.77 spot, 22 DTE, 12.9\% implied volatility.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \scriptsize
      \begin{block}{The short-volatility trade}
        Investors attempt to capture the volatility risk premium by selling options and accepting exposure to volatility and potentially severe market movements. The strategy can generate consistent premiums during relatively stable markets, but may experience substantial losses during periods of extreme volatility.
      \end{block}
      \begin{block}{Four legs, one credit}
        Buy 762P / sell 767P; sell 777C / buy 782C. Both wings are \$5 wide. Net credit and max profit: \$3.78 per share, or \textbf{\$378 per 1-lot}. Max loss: \$122.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.08cm}
  \scriptsize
  \begin{block}{At expiration}
    Max profit occurs when SPY finishes between the two short strikes (\$767--\$777). Beyond either long wing, the loss is capped at \$122 per 1-lot.
  \end{block}
\end{frame}
